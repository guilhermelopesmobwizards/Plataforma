CPA", data: plats.map(p => parseFloat(p.avg_cpa) || 0), backgroundColor: CHART_COLORS.slice(0, plats.length), borderWidth: 0, borderRadius: 4, }] }, options: { indexAxis: "y", responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { x: { grid: { color: "rgba(255,255,255,0.04)" }, ticks: { color: "#6b7280", font: { size: 10 } } }, y: { grid: { display: false }, ticks: { color: "#9ca3af", font: { size: 11 } } } } } }); // ROAS by client list const clients = data.by_client || []; document.getElementById("roas-list").innerHTML = clients.map((c, i) => { const roi = parseFloat(c.avg_roi); const color = roi >= 4 ? "var(--accent2)" : roi >= 3 ? "var(--warning)" : "var(--danger)"; const initials = (c.client_name || "?").split(" ").map(w => w[0]).join("").slice(0, 2).toUpperCase(); return `
${initials}
${c.client_name}
${c.campaigns} campanha${c.campaigns !== 1 ? "s" : ""}
${roi.toFixed(2)}×
ROAS
`; }).join(""); } window.addEventListener("DOMContentLoaded", loadMetrics);
